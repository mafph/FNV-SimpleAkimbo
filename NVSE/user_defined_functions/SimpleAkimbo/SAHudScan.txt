Begin Function {}
	; Debug helper: searches the HUDMainMenu for all tiles with non-empty text and prints path and text to the console.
	; Goal: find out in which tile the ammo type is shown. Changes nothing.
	; Call from console:  Call (CompileScript "SimpleAkimbo\SAHudScan.txt")
	; Call with Akimbo off and a known weapon equipped, then search the output for the ammo name.
	string_var sPath
	string_var sText
	int iA
	int iB
	int iLen

	; Level 1: HUDMainMenu\*:A\string
	let iA = 0
	while iA < 40
		let sPath = sv_Construct "HUDMainMenu\*:%g\string" iA
		let sText = GetUIString (sPath)
		let iLen = sv_Length sText
		if iLen > 0
			PrintC "SA HudScan %z = %z" sPath sText
		endif
		let iA += 1
	loop

	; Level 2: HUDMainMenu\*:A\*:B\string
	let iA = 0
	while iA < 40
		let iB = 0
		while iB < 25
			let sPath = sv_Construct "HUDMainMenu\*:%g\*:%g\string" iA iB
			let sText = GetUIString (sPath)
			let iLen = sv_Length sText
			if iLen > 0
				PrintC "SA HudScan %z = %z" sPath sText
			endif
			let iB += 1
		loop
		let iA += 1
	loop

	; Level 3: HUDMainMenu\*:A\*:B\*:C\string, up to 15 per level only
	let iA = 0
	while iA < 40
		let iB = 0
		while iB < 15
			let sPath = sv_Construct "HUDMainMenu\*:%g\*:%g\*:0\string" iA iB
			let sText = GetUIString (sPath)
			let iLen = sv_Length sText
			if iLen > 0
				PrintC "SA HudScan %z = %z" sPath sText
			endif
			let iB += 1
		loop
		let iA += 1
	loop
	PrintC "SA HudScan done"
End
