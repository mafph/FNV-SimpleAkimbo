Begin Function {int iKey}
	; Mouse button held: remember the state (counterpart SAHeldUp) and, if the hand is automatic, register automatic fire (SAAuto).
	; Also checks here whether the right weapon has changed since the last interval calculation (no frame comparison needed).
	ref rEquip
	ref rLast
	float fMode
	float fInt
	float fIdle

	if iKey == 256
		Player.AuxVarSetFlt "*_SAHeld256" 1
	else
		Player.AuxVarSetFlt "*_SAHeld257" 1
	endif
	if GetActiveMenuMode != 0
		return
	endif
	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		return
	endif

	let rEquip = Player.GetEquippedObject 5
	let rLast = Player.AuxVarGetRef "*_SAAutoRefR"
	if rEquip != rLast
		Call (CompileScript "SimpleAkimbo\SAAutoInit.txt")
	endif

	let fMode = Player.AuxVarGetFlt "*_SAMode"
	let fInt = 0
	if fMode == 0
		if iKey == 257
			let fInt = Player.AuxVarGetFlt "*_SAAutoInt1"
		else
			let fInt = Player.AuxVarGetFlt "*_SAAutoInt0"
		endif
	elseif iKey == 256
		let fInt = Player.AuxVarGetFlt "*_SAAutoIntB"
	endif
	if fInt > 0
		; first automatic shot comes after the interval plus 0.1 s, a short click gives one shot
		let fIdle = fInt + 0.1
		if iKey == 257
			Player.AuxVarSetFlt "*_SAAutoT1" fIdle
		else
			Player.AuxVarSetFlt "*_SAAutoT0" fIdle
		endif
		; unregister first, then register: never registered twice
		SetGameMainLoopCallback (CompileScript "SimpleAkimbo\SAAuto.txt") 0 1 1
		SetGameMainLoopCallback (CompileScript "SimpleAkimbo\SAAuto.txt") 1 1 1
	endif
End
