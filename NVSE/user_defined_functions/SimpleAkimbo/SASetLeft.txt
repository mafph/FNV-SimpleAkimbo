Begin Function {int iKey}
	; Called by J in the inventory (Pip-Boy, menu 1002). Remember the selected one-handed weapon as the left Akimbo weapon, without equipping it.
	; Akimbo off and a one-handed weapon equipped: Akimbo starts after the menu is closed.
	; Akimbo on and a different weapon selected: the left weapon is swapped after the menu is closed.
	; The equipped weapon must not be its own left weapon, unless there is a second copy.
	ref rWeap
	ref rItem
	ref rEquip
	ref rOld
	int iType
	int iWT
	int iEWT
	int iCount
	int iMods
	float fAbs

	if GetActiveMenuMode != 1002
		return
	endif

	let rItem = GetSelectedItemRef
	if (IsReference rItem) == 0
		PrintC "SimpleAkimbo: nothing selected, press J on a weapon in the inventory"
		return
	endif
	let rWeap = rItem.GetBaseObject
	let iType = GetType rWeap
	if iType != 40
		PrintC "SimpleAkimbo: selected item is not a weapon"
		return
	endif
	; weapon type restriction removed

	let rEquip = Player.GetEquippedObject 5
	if rWeap == rEquip
		let iCount = Player.GetItemCount rWeap
		if iCount < 2
			PrintC "SimpleAkimbo: the equipped weapon cannot be its own left weapon"
			return
		endif
	endif

	let iMods = rItem.GetWeaponRefModFlags
	; An already remembered left weapon is also set like a new one (no abort with a message)

	let fAbs = rItem.GetItemRefCurrentHealth 0
	; GetItemRefCurrentHealth 1 does not return a percentage, so the maximum comes from GetBaseHealth (SAWear)
	Player.AuxVarSetFlt "*_SALeftHP" fAbs
	Player.AuxVarSetFlt "*_SALeftMax" 0
	Player.AuxVarSetFlt "*_SAHPL" 0
	Player.AuxVarSetFlt "*_SACondL" 0
	Player.AuxVarSetRef "*_SALeftW" rWeap
	Player.AuxVarSetFlt "*_SALeftMods" iMods
	MessageEx "Akimbo left: %n" rWeap
	PrintC "SimpleAkimbo: left weapon = %i" rWeap
	; Log kind (0 other, 1 pistol, 2 revolver, 3 double barrel, 4 lever). Cylinder node only if Akimbo is on.
	let iWT = Call (CompileScript "SimpleAkimbo\SAWeaponKind.txt") rWeap 0
	PrintC "SimpleAkimbo: left weapon kind=%g" iWT

	if Player.AuxVarGetFlt "*_SAkimbo" == 1
		Player.AuxVarSetFlt "*_SASwapPend" 1
	elseif rEquip != 0
		Player.AuxVarSetFlt "*_SAStartPend" 1
	endif
End
