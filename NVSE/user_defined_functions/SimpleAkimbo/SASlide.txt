Begin Function {}
	; Slide control via script, main loop callback every frame (see SAStart).
	; 1) Hold-open: if the magazine of one hand is empty, the slide of that hand is held open.
	;    Only after the shot animation has finished (*_SAHoldDR/DL), otherwise the animation would be cut off.
	;    The value applies to the node itself. For weapons with adjusted nodes (SANodeFix) the rest position is 0.
	; 2) On the left, the slide does not always close fully after a shot (ActivateAnim): after
	;    the animation (*_SASlideTL) has finished, it is set to the rest position once.
	float fDt
	float fOpen
	float fAmmoR
	float fAmmoL
	float fT
	int iModel

	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		return
	endif
	let fDt = GetSecondsPassed
	let fOpen = 0 - 4.07
	let fAmmoR = Player.AuxVarGetFlt "*_SAAmmoR"
	let fAmmoL = Player.AuxVarGetFlt "*_SAAmmoL"

	if fAmmoR <= 0
		let fT = Player.AuxVarGetFlt "*_SAHoldDR"
		if fT > 0
			Player.AuxVarSetFlt "*_SAHoldDR" (fT - fDt)
		else
			let iModel = 1
			while iModel <= 2
				Player.SetNifBlockTranslation "##nmBar" fOpen 0 0 iModel
				let iModel += 1
			loop
		endif
	endif

	if fAmmoL <= 0
		let fT = Player.AuxVarGetFlt "*_SAHoldDL"
		if fT > 0
			Player.AuxVarSetFlt "*_SAHoldDL" (fT - fDt)
		else
			let iModel = 1
			while iModel <= 2
				Player.SetNifBlockTranslation "##nmBar1" fOpen 0 0 iModel
				let iModel += 1
			loop
		endif
	endif

	let fT = Player.AuxVarGetFlt "*_SASlideTL"
	if fT > 0
		let fT = fT - fDt
		Player.AuxVarSetFlt "*_SASlideTL" fT
		if fT <= 0 && fAmmoL > 0
			let iModel = 1
			while iModel <= 2
				Player.SetNifBlockTranslation "##nmBar1" 0 0 0 iModel
				let iModel += 1
			loop
		endif
	endif
End
