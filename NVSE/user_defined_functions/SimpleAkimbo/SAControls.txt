Begin Function {}
	; Block the engine attack (control 4) while Akimbo is on.
	; Mode A: additionally block Block/Aim (6), because RMB fires the right weapon.
	; Controls are NEVER blocked in menus (otherwise clicks in the menu would not work). If a menu is still open at call time
	; (e.g. close event of a Pip-Boy tab or a submenu in the container), it is checked again after 0.2 s
	; until the menu is really closed.
	float fOn
	float fMode
	let fOn = Player.AuxVarGetFlt "*_SAkimbo"
	let fMode = Player.AuxVarGetFlt "*_SAMode"
	EnableControl 4
	EnableControl 6
	EnableControl 5
	if GetActiveMenuMode != 0
		if fOn == 1
			Player.CallAfterSeconds 0.2 (CompileScript "SimpleAkimbo\SAControls.txt")
		endif
		return
	endif
	if fOn == 1
		DisableControl 4
		if fMode == 0
			DisableControl 6
		endif
	endif
End
