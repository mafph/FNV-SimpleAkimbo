Begin Function {int iForceOff}
	; Locomotion 1p fixed via ActivateAnim, priority 3 above the equip hold, below shot and reload.
	; Main loop callback every 6 frames, GameMode. iForceOff = 1: only remove the running sequence via SAAnimOff.
	; State: 0 = standing, 1 to 4 = walk forward, back, left, right, 5 to 8 = the same as running.
	; Direction from keys W 17, S 31, A 30, D 32, running = left Shift 42 held, not while sneaking. Default binding.
	; Sneak: the pack has no own sneak files for locomotion, sneaking uses the walk files.
	; 3p: the pack contains no locomotion files, only mtidle, therefore 1p only.
	string_var sBase
	string_var sName
	string_var sPath
	int iDir
	int iFast
	int iState
	int iOld
	int iPass
	int iSel
	int iFastSel
	int iLocoPrio

	; Priority = equip priority plus 1, so locomotion always lies above the equip hold
	let iLocoPrio = Player.AuxVarGetFlt "*_SAEquipPrio"
	if iLocoPrio <= 0
		let iLocoPrio = 2
	endif
	let iLocoPrio += 1

	let iOld = Player.AuxVarGetFlt "*_SALocoState"
	let iState = 0

	if iForceOff != 1
		if Player.AuxVarGetFlt "*_SAkimbo" != 1
			return
		endif
		if GetActiveMenuMode == 0
			if Player.IsWeaponOut == 1
				let iDir = 0
				if IsKeyPressed 17
					let iDir = 1
				elseif IsKeyPressed 31
					let iDir = 2
				elseif IsKeyPressed 30
					let iDir = 3
				elseif IsKeyPressed 32
					let iDir = 4
				endif
				if iDir > 0
					let iFast = 0
					if IsKeyPressed 42
						if Player.IsSneaking == 0
							let iFast = 1
						endif
					endif
					let iState = iDir + 4 * iFast
				endif
			endif
		endif
	endif

	if iState == iOld
		return
	endif

	; Locomotion folder comes from the active pack (slot Loco, 1st person only)
	let sName = Player.AuxVarGetStr "*_SAP_Loco_D"
	let iPass = sv_Length sName
	if iPass == 0
		return
	endif
	let sBase = sv_Construct "AnimGroupOverride\%z\_1stPerson\Locomotion" sName
	let iPass = 0
	while iPass < 2
		if iPass == 0
			let iSel = iOld
		else
			let iSel = iState
		endif
		if iSel > 0
			let iFastSel = 0
			if iSel > 4
				let iFastSel = 1
				let iSel -= 4
			endif
			if iSel == 1
				let sName = "forward"
			elseif iSel == 2
				let sName = "backward"
			elseif iSel == 3
				let sName = "left"
			else
				let sName = "right"
			endif
			if iFastSel == 1
				let sName = "fast" + sName
			endif
			let sPath = sBase + "\1hp" + sName + ".kf"
			if iPass == 0
				Player.DeactivateAnim (sPath) 0.0
			else
				Player.ActivateAnim (sPath) 1 iLocoPrio 1 1.0 0.1 ""
			endif
		endif
		let iPass += 1
	loop

	Player.AuxVarSetFlt "*_SALocoState" iState
End
