Begin Function {int iKey}
	string_var sPath
	string_var sTarget
	ref rWeap
	float fState
	int iOK
	int iLen
	ref rAmmo
	ref rEquip
	int iCount
	int iMenu
	int iMods
	int iWT
	string_var sHudPath
	string_var sHudOrig

	if iKey != -1
		let iMenu = GetActiveMenuMode
		if iMenu != 0
			; J in the inventory remembers the selected weapon, nothing happens in other menus
			if iMenu == 1002
				Call (CompileScript "SimpleAkimbo\SASetLeft.txt") 0
			endif
			return
		endif
	endif

	let fState = Player.AuxVarGetFlt "*_SAkimbo"

	if fState == 0
		; The equipped weapon is checked first, so the warning also appears when no left weapon is set yet
		let rEquip = Player.GetEquippedObject 5
		if rEquip == 0
			PrintC "SimpleAkimbo: no weapon equipped"
			return
		endif
		; only one-handed weapons (type 3 and 4), same warning as when marking a two-handed weapon with J in the inventory
		let iWT = GetWeaponType rEquip
		if iWT != 3 && iWT != 4
			PrintC "SimpleAkimbo: %n is not a one-handed weapon" rEquip
			MessageEx "%n is not a one-handed weapon" rEquip
			return
		endif
		; Model of the left weapon: remembered weapon (M), otherwise vanilla 9mm
		let rWeap = Player.AuxVarGetRef "*_SALeftW"
		if rWeap == 0
			PrintC "SimpleAkimbo: no left weapon set, press J on a weapon in the inventory"
			MessageEx "No left weapon set, press J on a weapon in the inventory"
			return
		endif
		let iWT = GetWeaponType rWeap
		if iWT != 3 && iWT != 4
			PrintC "SimpleAkimbo: left weapon is not one-handed"
			return
		endif
		if rWeap == rEquip
			let iCount = Player.GetItemCount rWeap
			if iCount < 2
				PrintC "SimpleAkimbo: left weapon is the equipped one, a second copy is required"
				return
			endif
		endif
		; Original text of the ammo line is in tile 2 (we write tile 1, where "WEAPON" was when read, tile 3 is "AP")
		let sHudPath = "HUDMainMenu\ActionPoints\justify_right_text:2\string"
		let sHudOrig = GetUIString (sHudPath)
		Player.AuxVarSetStr "*_SAHudOrig" (sHudOrig)
		Call (CompileScript "SimpleAkimbo\SAShellPurge.txt")
		let sPath = ""
		if rWeap != 0
			; Model with the mods of the selected copy (bitmask 0 to 7), otherwise base model
			let iMods = Player.AuxVarGetFlt "*_SALeftMods"
			let sPath = GetWeaponModel rWeap iMods
			let iLen = sv_Length sPath
			if iLen == 0 && iMods != 0
				let sPath = GetWeaponModel rWeap 0
			endif
		endif
		let iLen = sv_Length sPath
		if iLen == 0
			let sPath = "weapons\1handpistol\9mm.nif"
		endif
		PrintC "SimpleAkimbo: left model = %z" sPath
		Player.AuxVarSetStr "*_SAModel" (sPath)

		let sTarget = "Akimbo|*1*" + sPath
		; Adjust the node names of the right weapon to the animation scheme (before attaching the left one)
		Call (CompileScript "SimpleAkimbo\SANodeFix.txt") 1
		let iOK = InsertNode Player 3 "Weapon|Akimbo"
		if iOK == 0
			PrintC "SimpleAkimbo: InsertNode failed"
			return
		endif
		let iOK = AttachModel Player 3 (sTarget)
		if iOK == 0
			PrintC "SimpleAkimbo: AttachModel failed"
			InsertNode Player 2 "Weapon|Akimbo"
			return
		endif
		; Node names of the left weapon
		Call (CompileScript "SimpleAkimbo\SANodeFix.txt") 0
		Player.CallAfterSeconds 0.15 (CompileScript "SimpleAkimbo\SASlideRestore.txt")
		Player.AuxVarSetFlt "*_SAkimbo" 1
		Player.AuxVarSetFlt "*_SALPrio" 50
		Player.AuxVarSetFlt "*_SALastR" 0
		Player.AuxVarSetFlt "*_SASlideTL" 0
		Player.AuxVarSetFlt "*_SAAnimType" (Player.GetWeaponAnimType)
		Call (CompileScript "SimpleAkimbo\SAControls.txt")
		Player.AuxVarSetFlt "*_SAAmmoL" 0
		Player.AuxVarSetFlt "*_SAAmmoR" 0
		Player.AuxVarSetFlt "*_SACondR" 0
		Player.AuxVarSetFlt "*_SAHPR" 0
		Player.AuxVarSetFlt "*_SAJam" 0
		Player.AuxVarSetFlt "*_SAJamPend" 0
		Player.AuxVarSetFlt "*_SAAutoSeen" 0
		Call (CompileScript "SimpleAkimbo\SAAutoInit.txt")
		Call (CompileScript "SimpleAkimbo\SAReload.txt") 0
		PrintC "SimpleAkimbo: ON"
		Call (CompileScript "SimpleAkimbo\SAReq.txt") 1
	else
		let sPath = Player.AuxVarGetStr "*_SAModel"
		let sTarget = "Akimbo|*1*" + sPath
		PrintC "SimpleAkimbo: removing %z" sTarget
		Call (CompileScript "SimpleAkimbo\SAReq.txt") 0
		Player.AuxVarSetFlt "*_SAkimbo" 0
		Player.AuxVarSetFlt "*_SALPrio" 50
		Player.AuxVarSetFlt "*_SALastR" 0
		Player.AuxVarSetFlt "*_SASlideTL" 0
		SetGameMainLoopCallback (CompileScript "SimpleAkimbo\SAAuto.txt") 0 1 1
		Call (CompileScript "SimpleAkimbo\SAControls.txt")
		AttachModel Player 2 (sTarget)
		InsertNode Player 2 "Weapon|Akimbo"
		Player.AuxVarSetFlt "*_SAJam" 0
		Player.AuxVarSetFlt "*_SAJamPend" 0
		let sHudPath = "HUDMainMenu\ActionPoints\justify_right_text:1\string"
		let sHudOrig = Player.AuxVarGetStr "*_SAHudOrig"
		SetUIString (sHudPath) (sHudOrig)
		let rWeap = Player.GetEquippedObject 5
		if rWeap != 0
			let rAmmo = GetWeaponAmmo rWeap
			if rAmmo != 0
				Player.AddItem rAmmo 1 1
				Player.RemoveItem rAmmo 1 1
			endif
		endif
		PrintC "SimpleAkimbo: OFF"
		; Muzzle flash otherwise hangs on the left node: briefly unequip and re-equip the weapon (not when swapping the left weapon)
		if Player.AuxVarGetFlt "*_SASwapping" != 1
			let rWeap = Player.GetEquippedObject 5
			if rWeap != 0
				Player.AuxVarSetRef "*_SAReEquipW" rWeap
				Player.UnequipItem rWeap 0 1
				Player.CallAfterSeconds 0.2 (CompileScript "SimpleAkimbo\SAReEquip.txt")
			endif
		endif
	endif

	Player.RefreshAnimData
	; Equip animation when turning on and off. When turning off it comes from the re-equip (SAReEquip).
	if fState == 0
		Call (CompileScript "SimpleAkimbo\SAEquipAnim.txt")
		Player.AuxVarSetFlt "*_SASwapping" 0
	endif
End
