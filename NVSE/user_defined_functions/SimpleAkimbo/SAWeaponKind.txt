Begin Function {ref rWeap, int iSide}
	; Weapon class for the animation pack choice (SAPackApply).
	; iSide: 0 = left weapon (subtree "Akimbo"), 1 = right weapon (subtree "Weapon" minus "Akimbo")
	; Returns: 0 = none / other / unsupported, 1 = pistol, 2 = revolver, 3 = double barrel (sawed off), 4 = lever action
	; 1/2: GetWeaponType 3 (OneHandPistol) or 4 (OneHandPistolEnergy); node name containing "cylinder" -> revolver.
	;      The node check needs the weapon in the player's 3D (right weapon: equipped, left weapon: Akimbo ON).
	; 3/4: by substring of the weapon's model path. Pattern 3 = "sawedoff" (guess from the pack's model name, check with *_SADebug),
	;      pattern 4 (lever action) still EMPTY = off: look up the model path with *_SADebug and enter it in the PATTERN line.
	int iAnim
	int iKind
	int iType
	int iModel
	int iCylTotal
	int iCylAkimbo
	int iCylFound
	int iLen
	int iPos
	string_var sNodeName
	string_var sModel
	string_var sPat3
	string_var sPat4
	array_var aNodes
	array_var iter

	let iKind = 0
	if rWeap == 0
		SetFunctionValue 0
		return
	endif

	let iType = GetType rWeap
	if iType != 40
		SetFunctionValue 0
		return
	endif

	let iAnim = GetWeaponType rWeap
	if iAnim == 3 || iAnim == 4
		let iKind = 1

		let iCylTotal = 0
		let iCylAkimbo = 0
		let iModel = 1
		while iModel <= 2
			if iSide == 1
				let aNodes := Player.GetChildBlocks "Weapon" iModel
				if aNodes
					foreach iter <- aNodes
						let sNodeName := iter["value"]
						if Player.AuxVarGetFlt "*_SADebugNodes" == 1
							PrintC "SA Node: Weapon model=%g %z" iModel sNodeName
						endif
						if sv_Find "cylinder" sNodeName > -1
							let iCylTotal += 1
						endif
					loop
				endif
			endif
			let aNodes := Player.GetChildBlocks "Akimbo" iModel
			if aNodes
				foreach iter <- aNodes
					let sNodeName := iter["value"]
					if Player.AuxVarGetFlt "*_SADebugNodes" == 1
						PrintC "SA Node: Akimbo model=%g %z" iModel sNodeName
					endif
					if sv_Find "cylinder" sNodeName > -1
						let iCylAkimbo += 1
					endif
				loop
			endif
			let iModel += 1
		loop

		if iSide == 1
			let iCylFound = iCylTotal - iCylAkimbo
		else
			let iCylFound = iCylAkimbo
		endif
		if iCylFound > 0
			let iKind = 2
		endif
	else
		; PATTERN: lower-case part of the model path, e.g. "sawedoff"; empty = off
		let sPat3 = "sawedoff"
		let sPat4 = ""
		let sModel = GetWeaponModel rWeap 0
		if Player.AuxVarGetFlt "*_SADebug" == 1
			PrintC "SA WeaponKind model: %z" sModel
		endif
		let iLen = sv_Length sPat3
		if iLen > 0
			let iPos = sv_Find sPat3 sModel
			if iPos > -1
				let iKind = 3
			endif
		endif
		let iLen = sv_Length sPat4
		if iKind == 0 && iLen > 0
			let iPos = sv_Find sPat4 sModel
			if iPos > -1
				let iKind = 4
			endif
		endif
	endif

	if Player.AuxVarGetFlt "*_SADebug" == 1
		PrintC "SA WeaponKind: form=%i type=%g kind=%g side=%g (0 other 1 pistol 2 revolver 3 double 4 lever)" rWeap iAnim iKind iSide
	endif

	SetFunctionValue iKind
End
