Begin Function {}
	; Equip animation (1p + 3p). Clears holster flags.
	; *_SAEquipPhase = 1 while equip plays so stale CallAfterSeconds IdleHold cannot kill it.
	string_var sEquip1st
	string_var sEquip3rd
	ref rEquipNow
	int iPrio
	float fEquipLen
	float fGuard

	Player.AuxVarSetFlt "*_SAHolstered" 0
	Player.AuxVarSetFlt "*_SAHolsterCnt" 0
	Player.AuxVarSetFlt "*_SAHolsterQueued" 0
	Player.AuxVarSetFlt "*_SAHolsterLock" 0

	; Block IdleHold until equip duration elapsed (invalidates old timers)
	Player.AuxVarSetFlt "*_SAEquipPhase" 1

	let sEquip1st = "AnimGroupOverride\SA_Januusz9mm\_1stPerson\1hpEquip.kf"
	let sEquip3rd = "AnimGroupOverride\SA_Januusz9mm\_Male\1hpEquip.kf"
	let iPrio = Player.AuxVarGetFlt "*_SAEquipPrio"
	if iPrio <= 0
		let iPrio = 4
	endif

	Player.DeactivateAnim (sEquip1st) 0.0
	Player.DeactivateAnim (sEquip3rd) 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_1stPerson\1hpAttackLeft.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_Male\1hpAttackLeft.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_HandR\_1stPerson\1hpAttackRight.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_HandR\_Male\1hpAttackRight.kf" 0.0

	Player.ActivateAnim (sEquip1st) 1 iPrio 1 1.0 0.0 ""
	Player.ActivateAnim (sEquip3rd) 0 iPrio 1 1.0 0.0 ""

	let fEquipLen = Player.AuxVarGetFlt "*_SAEquipLen"
	if fEquipLen < 0.8
		let fEquipLen = 1.2
	endif
	; Only transition to idle hold after equip finished (via SAEquipToIdle - respects phase)
	Player.CallAfterSeconds fEquipLen (CompileScript "SimpleAkimbo\SAEquipToIdle.txt")
	Player.CallAfterSeconds (fEquipLen + 1.0) (CompileScript "SimpleAkimbo\SAEquipToIdle.txt")
	PrintC "SA Equip: anim started prio %g, hold in %g s" iPrio fEquipLen

	Player.AuxVarSetFlt "*_SAEquipGuard" 1
	let fGuard = fEquipLen + 0.8
	Player.CallAfterSeconds fGuard (CompileScript "SimpleAkimbo\SAEquipGuardEnd.txt")

	let rEquipNow = Player.GetEquippedObject 5
	Player.AuxVarSetRef "*_SAEquipLast" rEquipNow
End
