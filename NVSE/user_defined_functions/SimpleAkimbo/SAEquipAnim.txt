Begin Function {}
	; Equip animation fixed via ActivateAnim 1p and 3p. Also serves as an idle override:
	; the sequence stays CLAMP active and holds the last pose until SAAnimOff removes it.
	; Priority 2: below locomotion 3, shot 5 and reload 6. Changeable via *_SAEquipPrio.
	; Called: SAToggle when turning on, and SAEquipWatch when the right weapon changes.
	string_var sEquip1st
	string_var sEquip3rd
	ref rEquipNow
	int iPrio
	float fEquipLen
	float fGuard

	let sEquip1st = "AnimGroupOverride\SA_Januusz9mm\_1stPerson\1hpEquip.kf"
	let sEquip3rd = "AnimGroupOverride\SA_Januusz9mm\_Male\1hpEquip.kf"
	let iPrio = Player.AuxVarGetFlt "*_SAEquipPrio"
	if iPrio <= 0
		let iPrio = 2
	endif

	Player.DeactivateAnim (sEquip1st) 0.0
	Player.DeactivateAnim (sEquip3rd) 0.0
	Player.ActivateAnim (sEquip1st) 1 iPrio 1 1.0 0.0 ""
	Player.ActivateAnim (sEquip3rd) 0 iPrio 1 1.0 0.0 ""

	; After the equip time, SAIdleHold switches to the end pose of the attack kfs. Length via *_SAEquipLen, default 1.2 s.
	let fEquipLen = Player.AuxVarGetFlt "*_SAEquipLen"
	if fEquipLen <= 0
		let fEquipLen = 1.2
	endif
	; Additionally early and late, in case the engine removes the sequences again after the equip.
	Player.CallAfterSeconds 0.4 (CompileScript "SimpleAkimbo\SAIdleHold.txt")
	Player.CallAfterSeconds fEquipLen (CompileScript "SimpleAkimbo\SAIdleHold.txt")
	Player.CallAfterSeconds 2.5 (CompileScript "SimpleAkimbo\SAIdleHold.txt")
	PrintC "SA Equip: anim started, hold in 0.4 / %g / 2.5 s" fEquipLen

	; Equip phase: SATypeWatch must not replace the equip animation with a reload animation (ammo type changes with the weapon)
	Player.AuxVarSetFlt "*_SAEquipGuard" 1
	let fGuard = fEquipLen + 0.8
	Player.CallAfterSeconds fGuard (CompileScript "SimpleAkimbo\SAEquipGuardEnd.txt")

	let rEquipNow = Player.GetEquippedObject 5
	Player.AuxVarSetRef "*_SAEquipLast" rEquipNow
End
