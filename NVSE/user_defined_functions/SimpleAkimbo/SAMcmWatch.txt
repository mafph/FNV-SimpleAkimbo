Begin Function {}
	; Applies MCM changes without relying on what the MCM passes to a call script.
	; Runs as a main loop callback (GameMode), see SAStart. Changes take effect shortly after the MCM is closed.
	float fMode
	float fSeen
	float fStr
	float fSkill
	float fStrSeen
	float fSkSeen
	float fDebug
	int iOn
	int iAvR
	int iAvL
	int iNowR
	int iNowL

	; Debug messages
	let fDebug = Player.AuxVarGetFlt "*_SA_MCM" 3
	let fSeen = Player.AuxVarGetFlt "*_SADbgSeen"
	if fDebug != fSeen
		Player.AuxVarSetFlt "*_SADbgSeen" fDebug
		Player.AuxVarSetFlt "*_SADebug" fDebug
	endif

	; Fire mode
	let fMode = Player.AuxVarGetFlt "*_SA_MCM" 2
	let fSeen = Player.AuxVarGetFlt "*_SAMode"
	if fMode != fSeen
		Player.AuxVarSetFlt "*_SAMode" fMode
		Call (CompileScript "SimpleAkimbo\SAControls.txt")
		if fDebug == 1
			PrintC "SimpleAkimbo MCM: fire mode set to %g" fMode
		endif
	endif

	; Strength and skill requirement
	let fStr = Player.AuxVarGetFlt "*_SA_MCM" 4
	let fSkill = Player.AuxVarGetFlt "*_SA_MCM" 5
	let fStrSeen = Player.AuxVarGetFlt "*_SAStrMode"
	let fSkSeen = Player.AuxVarGetFlt "*_SASkMode"
	if fStr != fStrSeen || fSkill != fSkSeen
		Player.AuxVarSetFlt "*_SAStrMode" fStr
		Player.AuxVarSetFlt "*_SASkMode" fSkill
		let iOn = Player.AuxVarGetFlt "*_SAkimbo"
		Call (CompileScript "SimpleAkimbo\SAReq.txt") iOn
	endif

	; Weapons with different skills: the converted skill requirement depends on the current skill values
	if Player.AuxVarGetFlt "*_SAkimbo" == 1
		let iAvR = Player.AuxVarGetFlt "*_SAReqAvR"
		if iAvR > 0
			let iAvL = Player.AuxVarGetFlt "*_SAReqAvL"
			let iNowR = Player.GetAV iAvR
			let iNowL = Player.GetAV iAvL
			if iNowR != Player.AuxVarGetFlt "*_SAReqValR" || iNowL != Player.AuxVarGetFlt "*_SAReqValL"
				Call (CompileScript "SimpleAkimbo\SAReq.txt") 1
			endif
		endif
	endif
End
