Begin Function {}
	; Idle hold: after equip and after every reload, hold the end pose of the attack kfs.
	; StartOver 0: a sequence that is still active is not restarted.
	int iPrio

	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		return
	endif
	if Player.AuxVarGetFlt "*_SAHolstered" == 1
		return
	endif
	; Equip still playing - ignore stale CallAfterSeconds from older equip calls
	if Player.AuxVarGetFlt "*_SAEquipPhase" == 1
		PrintC "SA IdleHold: skipped, equip phase active"
		return
	endif
	if Player.AuxVarGetFlt "*_SAReloading" == 1
		PrintC "SA IdleHold: skipped, reload running"
		return
	endif

	let iPrio = Player.AuxVarGetFlt "*_SAEquipPrio"
	if iPrio <= 0
		let iPrio = 2
	endif

	PrintC "SA IdleHold: applied prio %g" iPrio
	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_1stPerson\1hpEquip.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_Male\1hpEquip.kf" 0.0

	Player.ActivateAnim "AnimGroupOverride\SA_Januusz9mm\_1stPerson\1hpAttackLeft.kf" 1 iPrio 0 1.0 0.0 ""
	Player.ActivateAnim "AnimGroupOverride\SA_Januusz9mm\_Male\1hpAttackLeft.kf" 0 iPrio 0 1.0 0.0 ""
	Player.ActivateAnim "AnimGroupOverride\SA_HandR\_1stPerson\1hpAttackRight.kf" 1 iPrio 0 1.0 0.0 ""
	Player.ActivateAnim "AnimGroupOverride\SA_HandR\_Male\1hpAttackRight.kf" 0 iPrio 0 1.0 0.0 ""
End
