Begin Function {int iKey}
	float fMode
	float fHand
	if GetActiveMenuMode != 0
		return
	endif
	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		return
	endif
	if Player.AuxVarGetFlt "*_SAEquipGuard" == 1
		return
	endif
	; Holstered: LMB always unholsters. RMB only if MCM UnholsterRMB (key 6).
	if Player.AuxVarGetFlt "*_SAHolstered" == 1
		if iKey == 256
			Call (CompileScript "SimpleAkimbo\SAUnholster.txt")
			return
		endif
		if iKey == 257 && Player.AuxVarGetFlt "*_SA_MCM" 6 == 1
			Call (CompileScript "SimpleAkimbo\SAUnholster.txt")
			return
		endif
		return
	endif
	let fMode = Player.AuxVarGetFlt "*_SAMode"
	if fMode == 0
		if iKey == 256
			Call (CompileScript "SimpleAkimbo\SAFire.txt") 0
		elseif iKey == 257
			Call (CompileScript "SimpleAkimbo\SAFire.txt") 1
		endif
	else
		if iKey == 256
			let fHand = Player.AuxVarGetFlt "*_SAHand"
			Call (CompileScript "SimpleAkimbo\SAFire.txt") (fHand)
			Player.AuxVarSetFlt "*_SAHand" (1 - fHand)
		endif
	endif
End
