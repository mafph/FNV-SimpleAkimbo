Begin Function {int iKey}
	float fMode
	float fHand
	if GetActiveMenuMode != 0
		return
	endif
	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		return
	endif
	let fMode = Player.AuxVarGetFlt "*_SAMode"
	if fMode == 0
		; Mode A: LMB left, RMB right
		if iKey == 256
			Call (CompileScript "SimpleAkimbo\SAFire.txt") 0
		elseif iKey == 257
			Call (CompileScript "SimpleAkimbo\SAFire.txt") 1
		endif
	else
		; Mode B: only LMB, hand alternates per shot, RMB stays engine ADS
		if iKey == 256
			let fHand = Player.AuxVarGetFlt "*_SAHand"
			Call (CompileScript "SimpleAkimbo\SAFire.txt") (fHand)
			Player.AuxVarSetFlt "*_SAHand" (1 - fHand)
		endif
	endif
End
