Begin Function {int iKey}
	; Mouse button released. If both buttons are up, automatic fire is unregistered.
	float fHeld256
	float fHeld257

	if iKey == 256
		Player.AuxVarSetFlt "*_SAHeld256" 0
	else
		Player.AuxVarSetFlt "*_SAHeld257" 0
	endif
	let fHeld256 = Player.AuxVarGetFlt "*_SAHeld256"
	let fHeld257 = Player.AuxVarGetFlt "*_SAHeld257"
	if fHeld256 == 0 && fHeld257 == 0
		SetGameMainLoopCallback (CompileScript "SimpleAkimbo\SAAuto.txt") 0 1 1
	endif
End
