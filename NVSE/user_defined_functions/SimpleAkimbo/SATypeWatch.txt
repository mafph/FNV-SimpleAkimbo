Begin Function {}
	; Detects the ammo change on the right (the engine itself triggers the vanilla reload)
	; and replaces it with the Akimbo reload. Runs as a main loop callback (GameMode), see SAStart.
	; While the reload animation is running (*_SATypeBusy), another change is only remembered
	; (*_SATypePend) and only applied silently after the animation ends (SATypeBusyEnd).
	ref rCurType
	ref rLastType
	ref rWeapNow
	ref rWeapLast

	let rCurType = Call (CompileScript "SimpleAkimbo\SAAmmoType.txt") 1
	let rLastType = Player.AuxVarGetRef "*_SATypeLast"
	let rWeapNow = Player.GetEquippedObject 5
	let rWeapLast = Player.AuxVarGetRef "*_SATypeWeap"

	; The right weapon itself changed: the ammo type changes with it. That is no type change on the same weapon,
	; so no reload animation here (the equip animation comes from SAWeaponSwap / SAToggle).
	if rWeapNow != rWeapLast
		Player.AuxVarSetRef "*_SATypeWeap" rWeapNow
		Player.AuxVarSetRef "*_SATypeLast" rCurType
		return
	endif

	if rLastType == rCurType
		return
	endif
	Player.AuxVarSetRef "*_SATypeLast" rCurType

	; Akimbo off or first run: only remember the type, trigger nothing
	if Player.AuxVarGetFlt "*_SAkimbo" != 1 || rLastType == 0
		return
	endif

	PrintC "SA TypeWatch ammo type changed %i -> %i" rLastType rCurType
	; During reload / type change animation: only swap the type silently, no second animation
	; (also during the equip phase and while a weapon swap is running)
	if Player.AuxVarGetFlt "*_SATypeBusy" == 1 || Player.AuxVarGetFlt "*_SAReloading" == 1 || Player.AuxVarGetFlt "*_SAEquipGuard" == 1 || Player.AuxVarGetFlt "*_SASwapBusy" == 1
		Call (CompileScript "SimpleAkimbo\SATypeApply.txt") 0
		return
	endif
	Call (CompileScript "SimpleAkimbo\SATypeApply.txt") 1
End
