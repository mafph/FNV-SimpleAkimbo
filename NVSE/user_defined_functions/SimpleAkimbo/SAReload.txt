Begin Function {int iKey}
	; iKey: 0 = still beide (Toggle), 19 = R beide / Auto, 44 = links, 45 = rechts
	float fLoaded
	float fCurL
	float fCurR
	float fPartial
	float fDur
	float fPOVInit
	string_var sAnim1st
	string_var sAnim3rd
	int iPrio

	if GetActiveMenuMode != 0
		PrintC "SimpleAkimbo: SAReload abgebrochen (Menue offen)"
		return
	endif
	Call (CompileScript "SimpleAkimbo\SACheck.txt")
	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		PrintC "SimpleAkimbo: SAReload abgebrochen (Akimbo aus)"
		return
	endif
	if Player.AuxVarGetFlt "*_SAJam" == 1
		PrintC "SimpleAkimbo: SAReload abgebrochen (Jam)"
		return
	endif
	if Player.AuxVarGetFlt "*_SAReloading" == 1
		return
	endif

	let fCurL = Player.AuxVarGetFlt "*_SAAmmoL"
	let fCurR = Player.AuxVarGetFlt "*_SAAmmoR"
	if iKey == 44
		let fPartial = (fCurL > 0)
	elseif iKey == 45
		let fPartial = (fCurR > 0)
	else
		let fPartial = (fCurL > 0 && fCurR > 0)
	endif

	Player.AuxVarSetFlt "*_SALoaded" 0
	Player.AuxVarSetFlt "*_SAJamPend" 0
	if iKey == 44
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 0
	elseif iKey == 45
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 1
	else
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 0
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 1
	endif

	Call (CompileScript "SimpleAkimbo\SAHud.txt")
	if iKey == 0
		return
	endif
	let fLoaded = Player.AuxVarGetFlt "*_SALoaded"
	if fLoaded <= 0
		PrintC "SimpleAkimbo: nothing to reload"
		return
	endif

	Player.AuxVarSetFlt "*_SAPartial" fPartial
	Call (CompileScript "SimpleAkimbo\SANodeFix.txt") 1
	Call (CompileScript "SimpleAkimbo\SANodeFix.txt") 0

	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_1stPerson\1hpAttackLeft.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_Male\1hpAttackLeft.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_1stPerson\1hpAttackLeftLast.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_Male\1hpAttackLeftLast.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_HandR\_1stPerson\1hpAttackRight.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_HandR\_Male\1hpAttackRight.kf" 0.0

	Player.SetNifBlockTranslation "##nmBar" 0 0 0 1
	Player.SetNifBlockTranslation "##nmBar" 0 0 0 2
	Player.SetNifBlockTranslation "##nmBar1" 0 0 0 1
	Player.SetNifBlockTranslation "##nmBar1" 0 0 0 2
	Player.AuxVarSetFlt "*_SALastR" 0
	Player.AuxVarSetFlt "*_SASlide3rdSeq" (Player.AuxVarGetFlt "*_SASlide3rdSeq" + 1)

	; Timer-Sperre
	; Sperrdauer = Laenge der Reload-kf (Full 5.05 s, Partial 4.42 s); per *_SAReloadLen ueberschreibbar
	let fDur = Player.AuxVarGetFlt "*_SAReloadLen"
	if fDur <= 0
		if fPartial == 1
			let fDur = 4.45
		else
			let fDur = 5.1
		endif
	endif
	Player.AuxVarSetFlt "*_SAReloading" 1
	Player.AuxVarSetFlt "*_SAReloadWait" 0
	Player.AuxVarSetFlt "*_SAReloadSeen" 0
	Player.AuxVarSetFlt "*_SAReloadSwitched" 0
	Player.AuxVarSetFlt "*_SAReloadDur" fDur
	let fPOVInit = IsPC1stPerson
	Player.AuxVarSetFlt "*_SAReloadPOV" fPOVInit

	; Reload per ActivateAnim mit festem Pfad (unabhaengig von waffenspezifischen Ordnern anderer Mods).
	; Das Deaktivieren am Ende passiert in SAReloadEnd.
	Player.DeactivateAnim "AnimGroupOverride\SA_ReloadFull\_1stPerson\1hpReloadL.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_ReloadFull\_Male\1hpReloadL.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_ReloadPartial\_1stPerson\1hpReloadL.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_ReloadPartial\_Male\1hpReloadL.kf" 0.0
	if fPartial == 1
		let sAnim1st = "AnimGroupOverride\SA_ReloadPartial\_1stPerson\1hpReloadL.kf"
		let sAnim3rd = "AnimGroupOverride\SA_ReloadPartial\_Male\1hpReloadL.kf"
	else
		let sAnim1st = "AnimGroupOverride\SA_ReloadFull\_1stPerson\1hpReloadL.kf"
		let sAnim3rd = "AnimGroupOverride\SA_ReloadFull\_Male\1hpReloadL.kf"
	endif
	let iPrio = 6
	Player.ActivateAnim (sAnim1st) 1 iPrio 1 1.0 0.0 ""
	Player.ActivateAnim (sAnim3rd) 0 iPrio 1 1.0 0.0 ""
	Player.CallAfterSeconds 0.05 (CompileScript "SimpleAkimbo\SAReloadEnd.txt")

	if iKey != 45
		Call (CompileScript "SimpleAkimbo\SAJam.txt") 0 1
	endif
	if iKey != 44
		Call (CompileScript "SimpleAkimbo\SAJam.txt") 1 1
	endif
End
