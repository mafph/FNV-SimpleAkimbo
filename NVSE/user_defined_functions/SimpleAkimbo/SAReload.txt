Begin Function {int iKey}
	; iKey: 0 = both silent (toggle), 19 = R both / auto, 44 = left, 45 = right
	float fLoaded
	float fCurL
	float fCurR
	float fPartial
	float fDur
	float fEnd
	float fCut
	float fPOVInit
	string_var sAnim1st
	string_var sAnim3rd
	array_var aAttr
	int iPrio
	int iDrop
	int iKL
	int iKR
	int iAttrSize
	float fAmmoLeftAfter
	float fAmmoRightAfter
	int iReloadNeedLeft
	int iReloadNeedRight
	ref rWL
	ref rWR

	if GetActiveMenuMode != 0
		PrintC "SimpleAkimbo: SAReload aborted (menu open)"
		return
	endif
	Call (CompileScript "SimpleAkimbo\SACheck.txt")
	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		PrintC "SimpleAkimbo: SAReload aborted (Akimbo off)"
		return
	endif
	; Equip guard must not block reload – only clear equip phase so IdleHold/TypeWatch settle
	if Player.AuxVarGetFlt "*_SAEquipGuard" == 1
		Player.AuxVarSetFlt "*_SAEquipPhase" 0
		Player.AuxVarSetFlt "*_SAEquipGuard" 0
		PrintC "SimpleAkimbo: SAReload – equip guard cleared"
	endif
	if Player.AuxVarGetFlt "*_SAHolstered" == 1
		PrintC "SimpleAkimbo: SAReload aborted (holstered)"
		return
	endif
	if Player.AuxVarGetFlt "*_SAJam" == 1
		PrintC "SimpleAkimbo: SAReload aborted (jam)"
		return
	endif
	if Player.AuxVarGetFlt "*_SAReloading" == 1
		return
	endif

	let fCurL = Player.AuxVarGetFlt "*_SAAmmoL"
	let fCurR = Player.AuxVarGetFlt "*_SAAmmoR"
	if iKey == 44
		let fPartial = (fCurL > 0)
	elseif iKey == 45
		let fPartial = (fCurR > 0)
	else
		let fPartial = (fCurL > 0 && fCurR > 0)
	endif

	Player.AuxVarSetFlt "*_SALoaded" 0
	Player.AuxVarSetFlt "*_SAJamPend" 0
	if iKey == 44
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 0
	elseif iKey == 45
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 1
	else
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 0
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 1
	endif

	Call (CompileScript "SimpleAkimbo\SAHud.txt")
	if iKey == 0
		return
	endif
	let fLoaded = Player.AuxVarGetFlt "*_SALoaded"
	if fLoaded <= 0
		PrintC "SimpleAkimbo: nothing to reload"
		return
	endif

	; Which hands actually received rounds? Only those take part in the single-hand sequence (SAReloadSeq).
	; Ammo type change (SATypeApply sets *_SAReloadForceBoth): both hands always take part.
	let fAmmoLeftAfter = Player.AuxVarGetFlt "*_SAAmmoL"
	let fAmmoRightAfter = Player.AuxVarGetFlt "*_SAAmmoR"
	let iReloadNeedLeft = 0
	let iReloadNeedRight = 0
	if fAmmoLeftAfter > fCurL
		let iReloadNeedLeft = 1
	endif
	if fAmmoRightAfter > fCurR
		let iReloadNeedRight = 1
	endif
	if Player.AuxVarGetFlt "*_SAReloadForceBoth" == 1
		let iReloadNeedLeft = 1
		let iReloadNeedRight = 1
		Player.AuxVarSetFlt "*_SAReloadForceBoth" 0
	endif
	Player.AuxVarSetFlt "*_SASeqNeedL" iReloadNeedLeft
	Player.AuxVarSetFlt "*_SASeqNeedR" iReloadNeedRight

	Player.AuxVarSetFlt "*_SAPartial" fPartial
	Call (CompileScript "SimpleAkimbo\SANodeFix.txt") 1
	Call (CompileScript "SimpleAkimbo\SANodeFix.txt") 0

	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_1stPerson\1hpAttackLeft.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_Male\1hpAttackLeft.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_1stPerson\1hpAttackLeftLast.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_Januusz9mm\_Male\1hpAttackLeftLast.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_HandR\_1stPerson\1hpAttackRight.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_HandR\_Male\1hpAttackRight.kf" 0.0

	Player.SetNifBlockTranslation "##nmBar" 0 0 0 1
	Player.SetNifBlockTranslation "##nmBar" 0 0 0 2
	Player.SetNifBlockTranslation "##nmBar1" 0 0 0 1
	Player.SetNifBlockTranslation "##nmBar1" 0 0 0 2
	Player.AuxVarSetFlt "*_SALastR" 0
	Player.AuxVarSetFlt "*_SASlide3rdSeq" (Player.AuxVarGetFlt "*_SASlide3rdSeq" + 1)

	; Weapon class per hand (SAWeaponKind): 1 = pistol (kf reload), 2 = other (single-hand sequence), 0 = none.
	; If at least one hand is "other", SAReloadSeq runs (each weapon reloads on its own). Console override: *_SAReloadKf 1 = always kf.
	let rWL = Player.AuxVarGetRef "*_SALeftW"
	let rWR = Player.GetEquippedWeapon
	let iKL = Call (CompileScript "SimpleAkimbo\SAWeaponKind.txt") rWL
	let iKR = Call (CompileScript "SimpleAkimbo\SAWeaponKind.txt") rWR
	let iDrop = 0
	if iKL == 2 || iKR == 2
		let iDrop = 1
	endif
	if Player.AuxVarGetFlt "*_SAReloadKf" == 1
		let iDrop = 0
	endif

	; Choose kf, then duration from GetAnimationAttributes endKeyTime (90% cut – feels less long).
	; Only used for pistols (iDrop == 0). Optional: *_SAReloadCut (default 0.9), *_SAReloadLen / Full / Partial absolute overrides.
	Player.DeactivateAnim "AnimGroupOverride\SA_ReloadFull\_1stPerson\1hpReloadL.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_ReloadFull\_Male\1hpReloadL.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_ReloadPartial\_1stPerson\1hpReloadL.kf" 0.0
	Player.DeactivateAnim "AnimGroupOverride\SA_ReloadPartial\_Male\1hpReloadL.kf" 0.0
	if fPartial == 1
		let sAnim1st = "AnimGroupOverride\SA_ReloadPartial\_1stPerson\1hpReloadL.kf"
		let sAnim3rd = "AnimGroupOverride\SA_ReloadPartial\_Male\1hpReloadL.kf"
		let fEnd = 4.417
	else
		let sAnim1st = "AnimGroupOverride\SA_ReloadFull\_1stPerson\1hpReloadL.kf"
		let sAnim3rd = "AnimGroupOverride\SA_ReloadFull\_Male\1hpReloadL.kf"
		let fEnd = 5.05
	endif
	; Real stop time from the kf if available (GetAnimationAttributes -> endKeyTime)
	let aAttr = GetAnimationAttributes sAnim1st
	let fEnd = 0
	let iAttrSize = ar_Size aAttr
	if iAttrSize > 0
		let fEnd = aAttr["endKeyTime"]
	endif
	if fEnd <= 0
		if fPartial == 1
			let fEnd = 4.417
		else
			let fEnd = 5.05
		endif
	endif
	let fCut = Player.AuxVarGetFlt "*_SAReloadCut"
	if fCut <= 0
		let fCut = 0.9
	endif
	if fCut > 1
		let fCut = 1
	endif
	let fDur = fEnd * fCut
	; Absolute overrides
	if fPartial == 1
		if Player.AuxVarGetFlt "*_SAReloadLenPartial" > 0
			let fDur = Player.AuxVarGetFlt "*_SAReloadLenPartial"
		endif
	else
		if Player.AuxVarGetFlt "*_SAReloadLenFull" > 0
			let fDur = Player.AuxVarGetFlt "*_SAReloadLenFull"
		endif
	endif
	if Player.AuxVarGetFlt "*_SAReloadLen" > 0
		let fDur = Player.AuxVarGetFlt "*_SAReloadLen"
	endif

	Player.AuxVarSetFlt "*_SAReloading" 1
	Player.AuxVarSetFlt "*_SAReloadWait" 0
	Player.AuxVarSetFlt "*_SAReloadSeen" 0
	Player.AuxVarSetFlt "*_SAReloadSwitched" 0
	Player.AuxVarSetFlt "*_SAReloadDur" fDur
	let fPOVInit = IsPC1stPerson
	Player.AuxVarSetFlt "*_SAReloadPOV" fPOVInit

	; Pistols (iDrop == 0): play the reload kf (prio 6), SAReloadEnd ends it.
	; Non-pistols (iDrop == 1): SAReloadSeq runs the single-hand sequence and ends the reload itself
	; (SAReloadEnd is NOT started, it would end the reload at once because Akimbo is off during the sequence).
	Player.AuxVarSetStr "*_SAReloadPath1st" (sAnim1st)
	Player.AuxVarSetStr "*_SAReloadPath3rd" (sAnim3rd)
	let iPrio = 6
	Player.AuxVarSetFlt "*_SAReloadHolster" 0
	if iDrop == 0
		Player.ActivateAnim (sAnim1st) 1 iPrio 1 1.0 0.0 ""
		Player.ActivateAnim (sAnim3rd) 0 iPrio 1 1.0 0.0 ""
		PrintC "SA Reload: partial=%g endKey=%g cut=%g dur=%g path=%z" fPartial fEnd fCut fDur sAnim1st
		Player.CallAfterSeconds 0.05 (CompileScript "SimpleAkimbo\SAReloadEnd.txt")
	else
		Player.AuxVarSetRef "*_SASeqRightW" rWR
		Player.AuxVarSetFlt "*_SASeqStep" 0
		PrintC "SA Reload: single-hand sequence"
		Call (CompileScript "SimpleAkimbo\SAReloadSeq.txt")
	endif

	if iKey != 45
		Call (CompileScript "SimpleAkimbo\SAJam.txt") 0 1
	endif
	if iKey != 44
		Call (CompileScript "SimpleAkimbo\SAJam.txt") 1 1
	endif
End
