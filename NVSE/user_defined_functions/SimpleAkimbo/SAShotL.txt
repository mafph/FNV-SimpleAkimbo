Begin Function {string_var sNode}
	; Left hand: remembered weapon (M), otherwise the equipped weapon as fallback.
	ref rWeap
	let rWeap = Player.AuxVarGetRef "*_SALeftW"
	if rWeap == 0
		PrintC "SimpleAkimbo: no left weapon set, using equipped weapon"
		Player.FireWeaponEx (Player.GetEquippedObject 5) (sNode)
		return
	endif
	if Player.AuxVarGetFlt "*_SADebug" == 1
		PrintC "SimpleAkimbo: left fires with form %i" rWeap
	endif
	Player.FireWeaponEx (rWeap) (sNode)
End
