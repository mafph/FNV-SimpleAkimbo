Begin Function {}
	; Debug helper: prints the text of justify_right_text:0 to :7 in the HUD (ActionPoints) to the console.
	; Call from console:  Call (CompileScript "SimpleAkimbo\SAHudDump.txt")
	; Run once with Akimbo off and once with Akimbo on and compare. Changes nothing.
	string_var sPath
	string_var sText
	int iTile
	int iLen

	let iTile = 0
	while iTile < 8
		let sPath = sv_Construct "HUDMainMenu\ActionPoints\justify_right_text:%g\string" iTile
		let sText = GetUIString (sPath)
		let iLen = sv_Length sText
		PrintC "SA HudDump tile %g len %g text %z" iTile iLen sText
		let iTile += 1
	loop
End
