Begin Function {int iHand}
	; Ammo type of a hand. 0 = left, 1 = right. Always returns a single ammo type.
	; GetWeaponAmmo returns an ammo list (FormList, e.g. AmmoList10mm) for many weapons.
	; GetItemCount on a list counts all types together, so the list is resolved here.
	; Right: type selected by the engine (GetPlayerCurrentAmmo).
	; Left: *_SAAmmoTypeL only if explicitly chosen (*_SAAmmoLSel = 1) and matching the left weapon.
	;        Otherwise left follows the right type if it matches the left weapon, otherwise the first type with stock.
	;        Nothing is saved automatically, so the left side does not stick to an old type.
	ref rAmmoType
	ref rBase
	ref rTypeWeap
	ref rEntry
	ref rRight
	int iIdx
	int iSize

	if iHand == 1
		let rAmmoType = GetPlayerCurrentAmmo
		if rAmmoType == 0
			let rTypeWeap = Player.GetEquippedObject 5
			let rAmmoType = GetWeaponAmmo rTypeWeap
			if GetType rAmmoType == 85
				let rAmmoType = ListGetNthForm rAmmoType 0
			endif
		endif
		SetFunctionValue rAmmoType
		return
	endif

	; Left
	let rTypeWeap = Player.AuxVarGetRef "*_SALeftW"
	if rTypeWeap == 0
		let rTypeWeap = Player.GetEquippedObject 5
	endif
	let rBase = GetWeaponAmmo rTypeWeap
	let rAmmoType = 0

	; 1) explicit selection on the left
	if Player.AuxVarGetFlt "*_SAAmmoLSel" == 1
		let rEntry = Player.AuxVarGetRef "*_SAAmmoTypeL"
		if rEntry != 0
			if GetType rBase == 85
				if (ListGetFormIndex rBase rEntry) >= 0
					let rAmmoType = rEntry
				endif
			elseif rEntry == rBase
				let rAmmoType = rEntry
			endif
		endif
	endif

	; 2) take the right type if it matches the left weapon
	if rAmmoType == 0
		let rRight = GetPlayerCurrentAmmo
		if rRight != 0
			if GetType rBase == 85
				if (ListGetFormIndex rBase rRight) >= 0
					let rAmmoType = rRight
				endif
			elseif rRight == rBase
				let rAmmoType = rRight
			endif
		endif
	endif

	; 3) first type in the list with stock
	if rAmmoType == 0
		if GetType rBase == 85
			let iSize = ListGetCount rBase
			let rAmmoType = ListGetNthForm rBase 0
			let iIdx = 0
			while iIdx < iSize
				let rEntry = ListGetNthForm rBase iIdx
				if Player.GetItemCount rEntry > 0
					let rAmmoType = rEntry
					break
				endif
				let iIdx += 1
			loop
		else
			let rAmmoType = rBase
		endif
	endif
	SetFunctionValue rAmmoType
End
