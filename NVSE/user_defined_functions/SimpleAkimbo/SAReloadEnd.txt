Begin Function {}
	; Reload lock ends after *_SAReloadDur (or immediately if Akimbo off).
	; Deactivates reload sequences, then either applies a queued holster or returns to idle hold.
	float fWait
	float fDur
	int iDone

	if Player.AuxVarGetFlt "*_SAReloading" != 1
		return
	endif

	let iDone = 0
	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		let iDone = 1
	else
		let fWait = Player.AuxVarGetFlt "*_SAReloadWait"
		let fWait += 0.05
		Player.AuxVarSetFlt "*_SAReloadWait" fWait
		let fDur = Player.AuxVarGetFlt "*_SAReloadDur"
		if fDur <= 0
			let fDur = 5.1
		endif
		if fWait >= fDur
			let iDone = 1
		endif
	endif

	if iDone == 1
		Player.AuxVarSetFlt "*_SAReloading" 0
		Player.AuxVarSetFlt "*_SAReloadWait" 0
		Player.DeactivateAnim "AnimGroupOverride\SA_ReloadFull\_1stPerson\1hpReloadL.kf" 0.0
		Player.DeactivateAnim "AnimGroupOverride\SA_ReloadFull\_Male\1hpReloadL.kf" 0.0
		Player.DeactivateAnim "AnimGroupOverride\SA_ReloadPartial\_1stPerson\1hpReloadL.kf" 0.0
		Player.DeactivateAnim "AnimGroupOverride\SA_ReloadPartial\_Male\1hpReloadL.kf" 0.0

		; Holster was requested mid-reload, or weapon is already away -> stay holstered
		if Player.AuxVarGetFlt "*_SAHolsterQueued" == 1 || Player.IsWeaponOut == 0
			Player.AuxVarSetFlt "*_SAHolsterQueued" 0
			Player.AuxVarSetFlt "*_SAHolstered" 1
			Player.AuxVarSetFlt "*_SAHolsterCnt" 2
			Player.AuxVarSetFlt "*_SAHolsterLock" 1
			Player.CallAfterSeconds 0.6 (CompileScript "SimpleAkimbo\SAHolsterLockEnd.txt")
			Call (CompileScript "SimpleAkimbo\SAAnimOff.txt")
			PrintC "SA ReloadEnd: holster applied after reload"
			return
		endif

		Call (CompileScript "SimpleAkimbo\SAIdleHold.txt")
		return
	endif

	Player.CallAfterSeconds 0.05 (CompileScript "SimpleAkimbo\SAReloadEnd.txt")
End
