Begin Function {int iMenuID}
	; Menu opened: release attack/aim so the mouse works in the menu.
	if Player.AuxVarGetFlt "*_SAkimbo" == 1
		EnableControl 4
		EnableControl 6
	endif
	; reset held mouse buttons, releasing one in the menu is not reliably reported
	Player.AuxVarSetFlt "*_SAHeld256" 0
	Player.AuxVarSetFlt "*_SAHeld257" 0
	SetGameMainLoopCallback (CompileScript "SimpleAkimbo\SAAuto.txt") 0 1 1
End
