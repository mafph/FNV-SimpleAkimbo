Begin Function {int iHand, int iReload}
	; Jam roll. iReload 0 = on shot, 1 = after reload.
	; Chance from the game values fWeaponConditionJam1 to 10 or fWeaponConditionReloadJam1 to 10,
	; so that changes made by mods apply. One value per 10 percent condition.
		; After reload: only set flag *_SAJamPend, SAFire reads it before the first shot and calls SAJamRun.
	; On shot: SAJamRun immediately. On jam: hide the left weapon (1 = 3rd person, 2 = 1st person), jam animation of the right weapon via ForceWeaponJamAnim, released after fJamSeconds.
	string_var sKey
	string_var sName
	float fPct
	float fChance
	int iBand
	int iRoll

	if Player.AuxVarGetFlt "*_SAJam" == 1
		return
	endif

	Call (CompileScript "SimpleAkimbo\SAWear.txt") iHand 0

	if iHand == 0
		let sKey = "*_SACondL"
	else
		let sKey = "*_SACondR"
	endif
	let fPct = Player.AuxVarGetFlt (sKey)
	let iBand = fPct / 10
	let iBand = iBand + 1
	if iBand > 10
		let iBand = 10
	endif
	if iBand < 1
		let iBand = 1
	endif

	if iReload == 1
		let sName = sv_Construct "fWeaponConditionReloadJam%g" iBand
	else
		let sName = sv_Construct "fWeaponConditionJam%g" iBand
	endif
	let fChance = GetNumericGameSetting (sName)
	let iRoll = GetRandomPercent
	if Player.AuxVarGetFlt "*_SADebug" == 1
		PrintC "SA Jam check hand %g, condition %g percent, band %g, chance %g, roll %g, after reload %g" iHand fPct iBand fChance iRoll iReload
	endif
	if iRoll >= fChance * 100
		return
	endif

	if Player.AuxVarGetFlt "*_SADebug" == 1
		PrintC "SimpleAkimbo: jam rolled hand %g, condition %g, chance %g, after reload %g" iHand fPct fChance iReload
	endif
	if iReload == 1
		Player.AuxVarSetFlt "*_SAJamPend" 1
	else
		Call (CompileScript "SimpleAkimbo\SAJamRun.txt")
	endif
End
