Begin Function {int iKey}
	; Key H: prints the texts of the child tiles justify_right_text:0..9,
	; so the ammo label can be found (e.g. "12/90").
	int i
	string_var sPath
	string_var sText
	if GetActiveMenuMode != 0
		return
	endif
	let i = 0
	while i < 10
		let sPath = sv_Construct "HUDMainMenu\ActionPoints\justify_right_text:%g\string" i
		let sText = GetUIString (sPath)
		PrintC "SimpleAkimbo HUD Tile %g = [%z]" i sText
		let i += 1
	loop
	PrintC "SimpleAkimbo: tile N with ammo found? See README item 2d"
End
