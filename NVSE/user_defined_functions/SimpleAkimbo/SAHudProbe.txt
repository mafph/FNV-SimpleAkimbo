Begin Function {int iKey}
	; Taste H: gibt die Texte der Kinder-Tiles justify_right_text:0..9 aus,
	; damit man das Ammo-Label findet (z.B. "12/90").
	int i
	string_var sPath
	string_var sText
	if GetActiveMenuMode != 0
		return
	endif
	let i = 0
	while i < 10
		let sPath = sv_Construct "HUDMainMenu\ActionPoints\justify_right_text:%g\string" i
		let sText = GetUIString (sPath)
		PrintC "SimpleAkimbo HUD Tile %g = [%z]" i sText
		let i += 1
	loop
	PrintC "SimpleAkimbo: Tile N mit Munition gefunden? Siehe LIES_MICH Punkt 2d"
End
