Begin Function {}
	; Ends Akimbo if the weapon animation type has changed since it was turned on
	; (e.g. switch to a rifle). The type is stored when turned on, no fixed number needed.
	float fType
	int iNow
	int iForce
	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		return
	endif
	let fType = Player.AuxVarGetFlt "*_SAAnimType"
	let iNow = Player.GetWeaponAnimType
	if iNow != fType
		PrintC "SimpleAkimbo: weapon type changed (%g -> %g), Akimbo off" fType iNow
		let iForce = 0 - 1
		Call (CompileScript "SimpleAkimbo\SAToggle.txt") iForce
	endif
End
