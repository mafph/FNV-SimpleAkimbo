Begin Function {}
	; Laeuft als Callback nur im Menue-Modus (siehe SAStart). Zeigt einen Hinweis mit dem Namen, wenn im Inventar eine Akimbo-Waffe markiert ist.
	ref rItem
	ref rBase
	ref rLast
	ref rLeft
	ref rEquip

	if GetActiveMenuMode != 1002
		return
	endif
	let rItem = GetSelectedItemRef
	if (IsReference rItem) == 0
		return
	endif
	let rBase = rItem.GetBaseObject
	let rLast = Player.AuxVarGetRef "*_SAHintLast"
	if rBase == rLast
		return
	endif
	Player.AuxVarSetRef "*_SAHintLast" rBase

	let rLeft = Player.AuxVarGetRef "*_SALeftW"
	let rEquip = Player.GetEquippedObject 5
	if rLeft != 0 && rBase == rLeft
		MessageEx "Akimbo left: %n" rBase
	elseif Player.AuxVarGetFlt "*_SAkimbo" == 1 && rBase == rEquip
		MessageEx "Akimbo right: %n" rBase
	endif
End
