Begin Function {int iHand, int iApply}
	; Suppressor sound. iApply 1 = before the shot: if a silencer or suppressor mod is active,
	; the weapon's attack sounds 0 to 2 are temporarily replaced by the mod sounds 9 to 11.
	; iApply 0 = after the shot: restore original sounds.
	ref rWeapon
	ref rSndOrig
	ref rSndMod
	string_var sSndKey
	int iModFlags
	int iSlotNum
	int iBitVal
	int iBitHit
	int iEffect
	int iSilenced
	int iSndType
	int iSndModId
	int iDidSwap

	if iApply == 0
		let rWeapon = Player.AuxVarGetRef "*_SASndW"
		if rWeapon == 0
			return
		endif
		let iSndType = 0
		while iSndType < 3
			let sSndKey = sv_Construct "*_SASndO%g" iSndType
			let rSndOrig = Player.AuxVarGetRef (sSndKey)
			if rSndOrig != 0
				SetWeaponSound rWeapon iSndType rSndOrig
			endif
			let iSndType += 1
		loop
		Player.AuxVarErase "*_SASndW"
		return
	endif

	; swap still active (delayed restore pending): do not back up anything twice
	let rWeapon = Player.AuxVarGetRef "*_SASndW"
	if rWeapon != 0
		return
	endif

	if iHand == 0
		let rWeapon = Player.AuxVarGetRef "*_SALeftW"
		if rWeapon == 0
			let rWeapon = Player.GetEquippedObject 5
		endif
		let iModFlags = Player.AuxVarGetFlt "*_SALeftMods"
	else
		let rWeapon = Player.GetEquippedObject 5
		let iModFlags = Player.GetEquippedWeaponModFlags
	endif
	if Player.AuxVarGetFlt "*_SADebug" == 1
		PrintC "SA Silence hand %g weapon %i mods %g" iHand rWeapon iModFlags
	endif
	if rWeapon == 0
		return
	endif
	if iModFlags == 0
		return
	endif

	let iSilenced = 0
	let iSlotNum = 1
	let iBitVal = 1
	while iSlotNum <= 3
		let iBitHit = iModFlags & iBitVal
		if iBitHit != 0
			let iEffect = GetWeaponItemModEffect iSlotNum rWeapon
			if iEffect == 11 || iEffect == 16
				let iSilenced = 1
			endif
		endif
		let iSlotNum += 1
		let iBitVal = iBitVal * 2
	loop
	if Player.AuxVarGetFlt "*_SADebug" == 1
		PrintC "SA Silence silenced %g" iSilenced
	endif
	if iSilenced == 0
		return
	endif

	let iDidSwap = 0
	let iSndType = 0
	while iSndType < 3
		let iSndModId = iSndType + 9
		let rSndMod = GetWeaponSound rWeapon iSndModId
		let rSndOrig = GetWeaponSound rWeapon iSndType
		if Player.AuxVarGetFlt "*_SADebug" == 1
			PrintC "SA Silence sound %g: original %i, mod %i" iSndType rSndOrig rSndMod
		endif
		if rSndOrig != 0
			let sSndKey = sv_Construct "*_SASndO%g" iSndType
			Player.AuxVarSetRef (sSndKey) rSndOrig
			if rSndMod != 0
				SetWeaponSound rWeapon iSndType rSndMod
			else
				; no mod sound stored (e.g. Distant): mute the sound entirely
				SetWeaponSound rWeapon iSndType
			endif
			let iDidSwap = 1
		endif
		let iSndType += 1
	loop
	if iDidSwap == 1
		Player.AuxVarSetRef "*_SASndW" rWeapon
	endif
End
