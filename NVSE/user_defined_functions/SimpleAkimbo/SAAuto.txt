Begin Function {}
	; Automatic fire. Only runs while LMB or RMB is held: SAHeldDown registers the callback, SAHeldUp, SAMenuOpen and SAToggle (off) unregister it.
	; At rest, automatic fire therefore costs no frame time. The first shot comes from the key press (SAInput).
	; Mode A: LMB left, RMB right, one timer each. Mode B: LMB, hand alternates per shot, one timer.
	string_var sIntKey
	string_var sTimerKey
	string_var sHeldKey
	float fDeltaTime
	float fInterval
	float fTimer
	float fModeVal
	float fJamVal
	float fHeldVal
	int iPass
	int iDoPass
	int iKeyCode

	if GetActiveMenuMode != 0
		return
	endif
	let fJamVal = Player.AuxVarGetFlt "*_SAJam"
	if fJamVal == 1
		; Jam: restart the timer, after release it only continues firing after a short pause
		Player.AuxVarSetFlt "*_SAAutoT0" 0.2
		Player.AuxVarSetFlt "*_SAAutoT1" 0.2
		return
	endif
	let fDeltaTime = GetSecondsPassed
	let fModeVal = Player.AuxVarGetFlt "*_SAMode"
	let iPass = 0
	while iPass < 2
		let iDoPass = 0
		if iPass == 0
			let iKeyCode = 256
			let sHeldKey = "*_SAHeld256"
			let sTimerKey = "*_SAAutoT0"
			if fModeVal == 0
				let sIntKey = "*_SAAutoInt0"
			else
				let sIntKey = "*_SAAutoIntB"
			endif
			let iDoPass = 1
		elseif fModeVal == 0
			let iKeyCode = 257
			let sHeldKey = "*_SAHeld257"
			let sTimerKey = "*_SAAutoT1"
			let sIntKey = "*_SAAutoInt1"
			let iDoPass = 1
		endif

		if iDoPass == 1
			let fHeldVal = Player.AuxVarGetFlt (sHeldKey)
			if fHeldVal == 1
				let fInterval = Player.AuxVarGetFlt (sIntKey)
				if fInterval > 0
					let fTimer = Player.AuxVarGetFlt (sTimerKey)
					let fTimer = fTimer - fDeltaTime
					if fTimer <= 0
						Player.AuxVarSetFlt (sTimerKey) fInterval
						Call (CompileScript "SimpleAkimbo\SAInput.txt") iKeyCode
					else
						Player.AuxVarSetFlt (sTimerKey) fTimer
					endif
				endif
			endif
		endif
		let iPass += 1
	loop
End
