Begin Function {int iHand}
	; iHand: 0 = left, 1 = right
	string_var sKeyAmmo
	string_var sNode
	string_var sAnim1st
	string_var sAnim3rd
	float fAmmo
	float fAmmoL
	float fAmmoR
	ref rFireAmmo
	int iPrio

	Call (CompileScript "SimpleAkimbo\SACheck.txt")
	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		return
	endif
	if Player.AuxVarGetFlt "*_SAEquipGuard" == 1
		return
	endif
	if Player.AuxVarGetFlt "*_SAHolstered" == 1
		return
	endif
	if Player.AuxVarGetFlt "*_SAJam" == 1
		return
	endif
	if Player.AuxVarGetFlt "*_SAReloading" == 1
		return
	endif

	if iHand == 0
		let sKeyAmmo = "*_SAAmmoL"
		let sNode = "ProjectileNode1"
	else
		let sKeyAmmo = "*_SAAmmoR"
		let sNode = "ProjectileNode"
	endif

	Call (CompileScript "SimpleAkimbo\SANodeRename.txt")
	let fAmmo = Player.AuxVarGetFlt (sKeyAmmo)
	if fAmmo <= 0
		let fAmmoL = Player.AuxVarGetFlt "*_SAAmmoL"
		let fAmmoR = Player.AuxVarGetFlt "*_SAAmmoR"
		if fAmmoL <= 0 && fAmmoR <= 0
			Call (CompileScript "SimpleAkimbo\SAReload.txt") 19
		else
			PrintC "SimpleAkimbo: magazine empty, press R to reload"
		endif
		return
	endif
	if Player.AuxVarGetFlt "*_SAJamPend" == 1
		Call (CompileScript "SimpleAkimbo\SAJamRun.txt")
		return
	endif
	let fAmmo -= 1
	Player.AuxVarSetFlt (sKeyAmmo) fAmmo

	; What is subtracted is exactly the type LOADED in this hand (single ammo, not a list).
	; GetWeaponAmmo often returns a FormList: RemoveItem on it subtracts from ALL types.
	if iHand == 0
		let rFireAmmo = Player.AuxVarGetRef "*_SALoadedTypeL"
	else
		let rFireAmmo = Player.AuxVarGetRef "*_SALoadedTypeR"
	endif
	if rFireAmmo == 0
		let rFireAmmo = Call (CompileScript "SimpleAkimbo\SAAmmoType.txt") iHand
	endif
	if rFireAmmo != 0
		Player.RemoveItem rFireAmmo 1 1
	endif

	if iHand == 0
		; Do NOT set HandSel here: switching to 0 aborts the running right animation (SA_HandR)

		Call (CompileScript "SimpleAkimbo\SAAnimStop.txt") "AtkL"
		Call (CompileScript "SimpleAkimbo\SAAnimStop.txt") "AtkLLast"

		; Fixed priority: 5 normal, 6 last shot
		let iPrio = 5

		if fAmmo == 0
			let sAnim1st = Call (CompileScript "SimpleAkimbo\SAAnim.txt") "AtkLLast" 1
			let sAnim3rd = Call (CompileScript "SimpleAkimbo\SAAnim.txt") "AtkLLast" 0
			let iPrio = 6
		else
			let sAnim1st = Call (CompileScript "SimpleAkimbo\SAAnim.txt") "AtkL" 1
			let sAnim3rd = Call (CompileScript "SimpleAkimbo\SAAnim.txt") "AtkL" 0
		endif

		Player.ActivateAnim (sAnim1st) 1 iPrio 1 1.0 0.0 ""
		Player.ActivateAnim (sAnim3rd) 0 iPrio 1 1.0 0.0 ""
		Player.AuxVarSetFlt "*_SALPrio" iPrio

		Call (CompileScript "SimpleAkimbo\SALeftSlide3rd.txt") fAmmo

		if fAmmo == 0
			Player.CallAfterSeconds 0.85 (CompileScript "SimpleAkimbo\SALeftHoldOpen.txt")
		else
			Player.CallAfterSeconds 0.85 (CompileScript "SimpleAkimbo\SALeftSlideClose.txt")
		endif
	else
		Player.AuxVarSetFlt "*_SAHandSel" 1

		; Right like left via ActivateAnim with a fixed path: independent of weapon-specific folders of other mods
		Call (CompileScript "SimpleAkimbo\SAAnimStop.txt") "AtkR"

		let iPrio = 5
		let sAnim1st = Call (CompileScript "SimpleAkimbo\SAAnim.txt") "AtkR" 1
		let sAnim3rd = Call (CompileScript "SimpleAkimbo\SAAnim.txt") "AtkR" 0
		Player.ActivateAnim (sAnim1st) 1 iPrio 1 1.0 0.0 ""
		Player.ActivateAnim (sAnim3rd) 0 iPrio 1 1.0 0.0 ""
	endif

	Call (CompileScript "SimpleAkimbo\SASilence.txt") iHand 1
	if iHand == 0
		Call (CompileScript "SimpleAkimbo\SAShotL.txt") (sNode)
	else
		Call (CompileScript "SimpleAkimbo\SAShot.txt") (sNode)
	endif
	Player.CallAfterSeconds 0.5 (CompileScript "SimpleAkimbo\SASilence.txt") 0 0 0

	Call (CompileScript "SimpleAkimbo\SARevolver.txt") iHand
	Call (CompileScript "SimpleAkimbo\SAShell.txt") iHand
	Call (CompileScript "SimpleAkimbo\SAHud.txt")
	Call (CompileScript "SimpleAkimbo\SAWear.txt") iHand 1
	Call (CompileScript "SimpleAkimbo\SAJam.txt") iHand 0

	let fAmmoL = Player.AuxVarGetFlt "*_SAAmmoL"
	let fAmmoR = Player.AuxVarGetFlt "*_SAAmmoR"
	if fAmmoL <= 0 && fAmmoR <= 0
		Player.CallAfterSeconds 0.15 (CompileScript "SimpleAkimbo\SAAutoReload.txt")
	endif

	if Player.AuxVarGetFlt "*_SADebug" == 1
		PrintC "SimpleAkimbo: shot hand=%g remaining=%g" iHand fAmmo
	endif
End
