Begin Function {}
	; Polls every 0.05 s and releases the type change lock.
	; In practice IsAnimPlaying ReloadL stays permanently 1, so it is not usable as an end signal.
	; End is therefore time-based: at the latest after length + 0.3 s (length from *_SAReloadLen, otherwise 2.5 s).
	; End earlier only if the animation was seen and afterwards returns 0 again.
	; If the change was made while a menu was open (*_SATypePendAnim), wait first
	; until the menu is closed, then play ReloadL and count the time from there.
	float fWait
	float fBusy
	float fLen
	float fSeen
	int iPlaying
	string_var sAnim1st
	string_var sAnim3rd
	int iPrio

	let fBusy = Player.AuxVarGetFlt "*_SATypeBusy"
	if fBusy != 1
		return
	endif

	if Player.AuxVarGetFlt "*_SATypePendAnim" == 1
		if GetActiveMenuMode != 0
			CallAfterSeconds 0.1 (CompileScript "SimpleAkimbo\SATypeBusyEnd.txt")
			return
		endif
		Player.AuxVarSetFlt "*_SATypePendAnim" 0
		Player.AuxVarSetFlt "*_SAPartial" 0
		Player.AuxVarSetFlt "*_SATypeWait" 0
		Player.AuxVarSetFlt "*_SATypeSeen" 0
		; Reload animation via ActivateAnim (full) incl. lock; end and deactivation in SAReloadEnd
	Call (CompileScript "SimpleAkimbo\SAAnimStop.txt") "RelFull"
	Call (CompileScript "SimpleAkimbo\SAAnimStop.txt") "RelPart"
		let sAnim1st = Call (CompileScript "SimpleAkimbo\SAAnim.txt") "RelFull" 1
		let sAnim3rd = Call (CompileScript "SimpleAkimbo\SAAnim.txt") "RelFull" 0
		let iPrio = 6
		Player.ActivateAnim (sAnim1st) 1 iPrio 1 1.0 0.0 ""
		Player.ActivateAnim (sAnim3rd) 0 iPrio 1 1.0 0.0 ""
		Player.AuxVarSetFlt "*_SAReloading" 1
		Player.AuxVarSetFlt "*_SAReloadWait" 0
		Player.AuxVarSetFlt "*_SAReloadDur" 5.1
		Player.CallAfterSeconds 0.05 (CompileScript "SimpleAkimbo\SAReloadEnd.txt")
		PrintC "SimpleAkimbo: Menu closed, reload animation started"
	endif

	let fWait = Player.AuxVarGetFlt "*_SATypeWait"
	let fWait = fWait + 0.05
	Player.AuxVarSetFlt "*_SATypeWait" fWait

	let fLen = Player.AuxVarGetFlt "*_SAReloadLen"
	if fLen <= 0
		let fLen = 2.5
	endif

	let iPlaying = Player.IsAnimPlaying ReloadL
	if iPlaying == 1
		Player.AuxVarSetFlt "*_SATypeSeen" 1
	endif
	let fSeen = Player.AuxVarGetFlt "*_SATypeSeen"

	PrintC "SimpleAkimbo: busy wait %g playing %g seen %g" fWait iPlaying fSeen

	if fWait < fLen + 0.3
		if fWait < 0.8 || fSeen == 0 || iPlaying == 1
			CallAfterSeconds 0.05 (CompileScript "SimpleAkimbo\SATypeBusyEnd.txt")
			return
		endif
	endif

	Player.AuxVarSetFlt "*_SATypeBusy" 0
	Player.AuxVarSetFlt "*_SATypeWait" 0
	Player.AuxVarSetFlt "*_SATypeSeen" 0
End
