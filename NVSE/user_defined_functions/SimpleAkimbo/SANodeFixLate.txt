Begin Function {}
	; Second NodeFix shortly after turning on: right after AttachModel the nodes are often not ready yet
	; (magazine/slide are then misplaced until the first reload). No argument, because CallAfterSeconds passes arguments unreliably.
	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		return
	endif
	Call (CompileScript "SimpleAkimbo\SANodeFix.txt") 1
	Call (CompileScript "SimpleAkimbo\SANodeFix.txt") 0
End
