Begin Function {int iApply}
	; Strength and skill requirement of the equipped (right) weapon while Akimbo is active.
	; Strength modes (MCM key 4): 0 = +0 (free), 1 = highest of both, 2 = highest +1, 3 = highest +50 percent of left, 4 = highest +100 percent of left.
	; Skill modes (MCM key 5): 0 = +0 (free), 1 = highest of both, 2 = highest +25, 3 = highest +100 percent of left.
	; The two weapons can use different skills (Guns / Energy Weapons ...) and the player can be skilled differently in them.
	; The engine only checks the right weapon against ITS skill. So the requirement of the left weapon is converted into the
	; skill of the right weapon: left requirement + (player value in right skill - player value in left skill).
	; The shortfall of the right weapon is then exactly the larger of the two real shortfalls.
	; iApply 1 = apply, 0 = only restore the original values.
	ref rOld
	ref rLeft
	ref rEquip
	int iMode
	int iOrig
	int iLeft
	int iBase
	int iNew
	int iSkR
	int iSkL
	int iValR
	int iValL
	int iLeftAdj

	Player.AuxVarSetFlt "*_SAReqAvR" 0
	Player.AuxVarSetFlt "*_SAReqAvL" 0
	; Restore the weapons changed before
	let rOld = Player.AuxVarGetRef "*_SAStrForm"
	if rOld != 0
		let iOrig = Player.AuxVarGetFlt "*_SAStrOrig"
		Player.SetReqStr iOrig rOld
		Player.AuxVarErase "*_SAStrForm"
	endif
	let rOld = Player.AuxVarGetRef "*_SASkForm"
	if rOld != 0
		let iOrig = Player.AuxVarGetFlt "*_SASkOrig"
		Player.SetReqSkill iOrig rOld
		Player.AuxVarErase "*_SASkForm"
	endif

	if iApply == 0
		return
	endif

	let rLeft = Player.AuxVarGetRef "*_SALeftW"
	let rEquip = Player.GetEquippedObject 5
	if rLeft == 0
		return
	endif
	if rEquip == 0
		return
	endif

	; Strength
	let iMode = Player.AuxVarGetFlt "*_SA_MCM" 4
	if iMode != 0
		let iOrig = Player.GetReqStr rEquip
		let iLeft = Player.GetReqStr rLeft
		let iBase = iOrig
		if iLeft > iBase
			let iBase = iLeft
		endif
		let iNew = iBase
		if iMode == 2
			let iNew = iBase + 1
		elseif iMode == 3
			let iNew = iBase + (iLeft / 2)
		elseif iMode == 4
			let iNew = iBase + iLeft
		endif
		Player.AuxVarSetFlt "*_SAStrOrig" iOrig
		Player.AuxVarSetRef "*_SAStrForm" rEquip
		Player.SetReqStr iNew rEquip
		if Player.AuxVarGetFlt "*_SADebug" == 1
			PrintC "SimpleAkimbo: strength req %g -> %g (mode %g)" iOrig iNew iMode
		endif
	endif

	; Skill
	let iMode = Player.AuxVarGetFlt "*_SA_MCM" 5
	if iMode != 0
		let iOrig = Player.GetReqSkill rEquip
		let iLeft = Player.GetReqSkill rLeft
		let iSkR = GetWeaponSkill rEquip
		let iSkL = GetWeaponSkill rLeft
		; same skill: left requirement unchanged. Different skills: convert into the skill of the right weapon.
		let iLeftAdj = iLeft
		if iSkR != iSkL && iSkR > 0 && iSkL > 0
			let iValR = Call (CompileScript "SimpleAkimbo\SASkillVal.txt") iSkR
			let iValL = Call (CompileScript "SimpleAkimbo\SASkillVal.txt") iSkL
			let iLeftAdj = iLeft + iValR - iValL
			; remember the skill values: SAMcmWatch recalculates when they change (chems, level up)
			Player.AuxVarSetFlt "*_SAReqAvR" iSkR
			Player.AuxVarSetFlt "*_SAReqAvL" iSkL
			Player.AuxVarSetFlt "*_SAReqValR" iValR
			Player.AuxVarSetFlt "*_SAReqValL" iValL
		endif
		let iBase = iOrig
		if iLeftAdj > iBase
			let iBase = iLeftAdj
		endif
		let iNew = iBase
		if iMode == 2
			let iNew = iBase + 25
		elseif iMode == 3
			let iNew = iBase + iLeft
		endif
		if iNew > 255
			let iNew = 255
		endif
		Player.AuxVarSetFlt "*_SASkOrig" iOrig
		Player.AuxVarSetRef "*_SASkForm" rEquip
		Player.SetReqSkill iNew rEquip
		if Player.AuxVarGetFlt "*_SADebug" == 1
			PrintC "SimpleAkimbo: skill req %g -> %g (mode %g, right skill %g, left skill %g, left req %g converted %g)" iOrig iNew iMode iSkR iSkL iLeft iLeftAdj
		endif
	endif
End
