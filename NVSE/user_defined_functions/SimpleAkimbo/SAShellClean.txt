Begin Function {}
	; Safety net: deletes shell casings that have been registered for longer than 6 seconds, e.g. after loading a save. Time via GameDaysPassed and TimeScale.
	; Runs as a main loop callback, see SAStart.
	ref rShell
	string_var sKey
	string_var sTimeKey
	int iHand
	int iSlot
	int iAlive
	float fStamp
	float fLife

	; only work when shell casings are registered (flag set by SAShell, cleared by SAShellPurge and this run)
	if Player.AuxVarGetFlt "*_SAShellActive" != 1
		return
	endif
	let iAlive = 0
	let fLife = 6 * TimeScale / 86400

	let iHand = 0
	while iHand < 2
		let iSlot = 0
		while iSlot < 60
			let sTimeKey = sv_Construct "*_SAShellT%g_%g" iHand iSlot
			let fStamp = Player.AuxVarGetFlt (sTimeKey)
			if fStamp != 0
				if GameDaysPassed - fStamp > fLife
					let sKey = sv_Construct "*_SAShell%g_%g" iHand iSlot
					let rShell = Player.AuxVarGetRef (sKey)
					if IsReference rShell
						rShell.Disable
						rShell.MarkForDelete
					endif
					Player.AuxVarSetFlt (sTimeKey) 0
				else
					let iAlive += 1
				endif
			endif
			let iSlot += 1
		loop
		let iHand += 1
	loop
	if iAlive == 0
		Player.AuxVarSetFlt "*_SAShellActive" 0
	endif
End
