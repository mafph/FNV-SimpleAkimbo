Begin Function {}
	; Main loop callback every 10 frames, GameMode (registered in SAStart).
	; Detects a change of the right weapon while Akimbo is running and hands over to SAWeaponSwap:
	; other one-handed weapon = Akimbo briefly off and on again (with equip animation), otherwise Akimbo off.
	; Without Akimbo the weapon is only remembered.
	ref rNow
	ref rLast

	let rNow = Player.GetEquippedObject 5
	let rLast = Player.AuxVarGetRef "*_SAEquipLast"

	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		Player.AuxVarSetRef "*_SAEquipLast" rNow
		return
	endif
	if rNow == rLast
		return
	endif
	if rLast == 0
		Player.AuxVarSetRef "*_SAEquipLast" rNow
		return
	endif
	if rNow == 0
		; weapon unequipped: wait for the next one
		return
	endif
	if Player.AuxVarGetFlt "*_SASwapBusy" == 1
		return
	endif
	if GetActiveMenuMode != 0
		return
	endif
	Call (CompileScript "SimpleAkimbo\SAWeaponSwap.txt")
End
