Begin Function {}
	; Main loop callback every 10 frames, GameMode: detects a change of the right weapon while Akimbo is running
	; and then plays the equip animation SAEquipAnim. Without Akimbo the weapon is only remembered.
	ref rNow
	ref rLast

	let rNow = Player.GetEquippedObject 5
	let rLast = Player.AuxVarGetRef "*_SAEquipLast"

	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		Player.AuxVarSetRef "*_SAEquipLast" rNow
		return
	endif
	if rNow == rLast
		return
	endif
	if rLast == 0
		Player.AuxVarSetRef "*_SAEquipLast" rNow
		return
	endif
	if rNow == 0
		return
	endif
	Call (CompileScript "SimpleAkimbo\SAEquipAnim.txt")
End
