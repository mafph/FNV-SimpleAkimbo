Begin Function {}
	; HUD ammo label: "left loaded/reserve   right loaded/reserve", e.g. 12/99  8/45.
	; Runs as a main loop callback (every 2 frames, GameMode only), see SAStart.
	; Reserve per hand from the hand's ammo type (SAAmmoType), not from the weapon default.
	ref rAmmoL
	ref rAmmoR
	float fTile
	float fMagL
	float fMagR
	int iStockL
	int iStockR
	string_var sPath
	string_var sText

	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		return
	endif

	let fTile = 1
	let fMagL = Player.AuxVarGetFlt "*_SAAmmoL"
	let fMagR = Player.AuxVarGetFlt "*_SAAmmoR"

	let rAmmoL = Call (CompileScript "SimpleAkimbo\SAAmmoType.txt") 0
	let rAmmoR = Call (CompileScript "SimpleAkimbo\SAAmmoType.txt") 1
	let iStockL = Player.GetItemCount rAmmoL
	let iStockR = Player.GetItemCount rAmmoR
	; Reserve = stock minus loaded rounds. Same type: subtract both magazines.
	if rAmmoL == rAmmoR
		let iStockL = iStockL - fMagL - fMagR
		let iStockR = iStockL
	else
		let iStockL = iStockL - fMagL
		let iStockR = iStockR - fMagR
	endif
	if iStockL < 0
		let iStockL = 0
	endif
	if iStockR < 0
		let iStockR = 0
	endif

	let sPath = sv_Construct "HUDMainMenu\ActionPoints\justify_right_text:%g\string" fTile
	let sText = sv_Construct "%g|%g %g|%g" fMagL iStockL fMagR iStockR
	SetUIString (sPath) (sText)
End
