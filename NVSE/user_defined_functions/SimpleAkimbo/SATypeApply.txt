Begin Function {int iWithAnim}
	; Apply ammo type change. iWithAnim 1 = with Akimbo reload animation (SAReload), 0 = silent (data only).
	; Magazines whose loaded type no longer matches the target type are set to 0
	; (the rounds count toward the reserve again) and are then refilled.
	; Left follows right (see SAAmmoType), so left is reloaded as well if needed.
	; If a menu is open, SAReload aborts (no animation): then only the data is
	; swapped and the animation is played via *_SATypePendAnim once the menu is closed.
	ref rAmmoNowR
	ref rAmmoNowL
	ref rAmmoLoadR
	ref rAmmoLoadL
	ref rNull
	float fLeftMagCount
	float fLoadedNow
	float fWasBusy
	int iHandRChanged
	int iHandLChanged

	let rAmmoNowR = Call (CompileScript "SimpleAkimbo\SAAmmoType.txt") 1
	let rAmmoNowL = Call (CompileScript "SimpleAkimbo\SAAmmoType.txt") 0
	let rAmmoLoadR = Player.AuxVarGetRef "*_SALoadedTypeR"
	let rAmmoLoadL = Player.AuxVarGetRef "*_SALoadedTypeL"
	let fLeftMagCount = Player.AuxVarGetFlt "*_SAAmmoL"

	let iHandRChanged = 0
	if rAmmoLoadR != rAmmoNowR
		let iHandRChanged = 1
	endif
	let iHandLChanged = 0
	if fLeftMagCount > 0
		if rAmmoLoadL != 0
			if rAmmoLoadL != rAmmoNowL
				let iHandLChanged = 1
			endif
		endif
	endif
	if iHandRChanged == 0
		if iHandLChanged == 0
			return
		endif
	endif

	; If a reload is currently running, the type is only swapped silently (no additional animation)
	if Player.AuxVarGetFlt "*_SAReloading" == 1
		let iWithAnim = 0
	endif

	if iHandRChanged == 1
		Player.AuxVarSetFlt "*_SAAmmoR" 0
		Player.AuxVarSetRef "*_SALoadedTypeR" rNull
	endif
	if iHandLChanged == 1
		Player.AuxVarSetFlt "*_SAAmmoL" 0
		Player.AuxVarSetRef "*_SALoadedTypeL" rNull
	endif

	; Silent (requested or menu open): only refill the data
	if iWithAnim == 0 || GetActiveMenuMode != 0
		if iHandRChanged == 1
			Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 1
		endif
		if iHandLChanged == 1
			Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 0
		endif
		Call (CompileScript "SimpleAkimbo\SAHud.txt")
		if iWithAnim == 1
			PrintC "SimpleAkimbo: Type change while menu open, animation follows after menu closes"
			Player.AuxVarSetFlt "*_SATypePendAnim" 1
			let fWasBusy = Player.AuxVarGetFlt "*_SATypeBusy"
			Player.AuxVarSetFlt "*_SATypeBusy" 1
			if fWasBusy != 1
				Player.AuxVarSetFlt "*_SATypeWait" 0
				Player.AuxVarSetFlt "*_SATypeSeen" 0
				CallAfterSeconds 0.1 (CompileScript "SimpleAkimbo\SATypeBusyEnd.txt")
			endif
		endif
		return
	endif

	; Always play the Akimbo animation (key 19: both hands, animation ReloadL).
	; SALoaded is set to 0 beforehand, so afterwards it shows whether SAReload actually loaded.
	Player.AuxVarSetFlt "*_SALoaded" 0
	Call (CompileScript "SimpleAkimbo\SAReload.txt") 19
	let fLoadedNow = Player.AuxVarGetFlt "*_SALoaded"
	if fLoadedNow <= 0
		; SAReload exited early (reload running, jam, menu ...) or there was no stock.
		; Magazines were set to 0 above, so refill silently here.
		PrintC "SimpleAkimbo: Type change, SAReload loaded nothing, refilling silently"
		if iHandRChanged == 1
			Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 1
		endif
		if iHandLChanged == 1
			Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 0
		endif
		Call (CompileScript "SimpleAkimbo\SAHud.txt")
		return
	endif

	; Busy flag only for the polling in SATypeBusyEnd (one chain, no blocking).
	; New change during the animation: reset the counter, the chain keeps running.
	let fWasBusy = Player.AuxVarGetFlt "*_SATypeBusy"
	Player.AuxVarSetFlt "*_SATypeBusy" 1
	Player.AuxVarSetFlt "*_SATypeWait" 0
	Player.AuxVarSetFlt "*_SATypeSeen" 0
	if fWasBusy != 1
		CallAfterSeconds 0.05 (CompileScript "SimpleAkimbo\SATypeBusyEnd.txt")
	endif
End
