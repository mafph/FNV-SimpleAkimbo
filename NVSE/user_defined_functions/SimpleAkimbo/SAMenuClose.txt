Begin Function {int iMenuID}
	; Menu closed: check for weapon change if needed, then set the lock again.
	Call (CompileScript "SimpleAkimbo\SACheck.txt")
	Call (CompileScript "SimpleAkimbo\SAControls.txt")
	Call (CompileScript "SimpleAkimbo\SAHud.txt")
	; J in the inventory selected a left weapon, start Akimbo shortly after closing
	if Player.AuxVarGetFlt "*_SAStartPend" == 1
		Player.AuxVarSetFlt "*_SAStartPend" 0
		Player.CallAfterSeconds 0.3 (CompileScript "SimpleAkimbo\SAToggle.txt") 0 0
		endif
	; J in the inventory selected a different left weapon while Akimbo is running: off and on again
	if Player.AuxVarGetFlt "*_SASwapPend" == 1
		Player.AuxVarSetFlt "*_SASwapPend" 0
		Player.AuxVarSetFlt "*_SASwapping" 1
		Player.CallAfterSeconds 0.3 (CompileScript "SimpleAkimbo\SAToggle.txt") 0 0
		Player.CallAfterSeconds 0.9 (CompileScript "SimpleAkimbo\SAToggle.txt") 0 0
	endif
	Player.AuxVarErase "*_SAHintLast"
End
