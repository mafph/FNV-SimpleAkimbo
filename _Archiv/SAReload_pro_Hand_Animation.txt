Begin Function {int iKey}
	; iKey: 0 = still beide (beim Einschalten), 19 = R beide, 44 = links, 45 = rechts
	; Scancodes sind Tastenpositionen (US-Layout): 44 = Z, 45 = X. Auf deutscher Tastatur sitzt 44 auf Y.
	float fLoaded

	if GetActiveMenuMode != 0
		return
	endif
	if Player.AuxVarGetFlt "*_SAkimbo" != 1
		return
	endif

	Player.AuxVarSetFlt "*_SALoaded" 0

	if iKey == 44
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 0
	elseif iKey == 45
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 1
	else
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 0
		Call (CompileScript "SimpleAkimbo\SAReloadHand.txt") 1
	endif

	if iKey == 0
		return
	endif

	let fLoaded = Player.AuxVarGetFlt "*_SALoaded"
	if fLoaded <= 0
		PrintC "SimpleAkimbo: nichts nachzuladen"
		return
	endif

	; Animation: beide = ReloadB, einzeln: 1st Person ReloadL/ReloadR, 3rd Person ReloadA
	if iKey == 44
		if IsPC1stPerson
			Player.PlayGroup ReloadL 1
		else
			Player.PlayGroup ReloadA 1
		endif
	elseif iKey == 45
		if IsPC1stPerson
			Player.PlayGroup ReloadR 1
		else
			Player.PlayGroup ReloadA 1
		endif
	else
		Player.PlayGroup ReloadB 1
	endif
End
